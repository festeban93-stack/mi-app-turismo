'use client';

import { useEffect, useState, useRef } from 'react';

interface Place {
  id: string;
  name: string;
  category: string;
  region: string;
  short_description: string;
  ai_story: string;
  tags: string[];
  estimated_price_usd: number;
  rating: number;
  location: any;
}

export default function InteractiveMap() {
  const [places, setPlaces] = useState<Place[]>([]);
  const [selectedRegion, setSelectedRegion] = useState('Todas');
  const [selectedCategory, setSelectedCategory] = useState('Todas');
  const [selectedPlace, setSelectedPlace] = useState<Place | null>(null);
  const [loading, setLoading] = useState(true);
  const [mounted, setMounted] = useState(false);
  
  const mapRef = useRef<any>(null);
  const markersRef = useRef<any[]>([]);

  const regions = ['Todas', 'San José', 'La Fortuna / San Carlos', 'Manuel Antonio', 'Guanacaste'];
  const categories = ['Todas', 'Naturaleza', 'Gastronomía', 'Aventura', 'Cultura'];

  // Asegurar renderizado solo en cliente
  useEffect(() => {
    setMounted(true);
  }, []);

  // Cargar CDN de Leaflet
  useEffect(() => {
    if (!mounted) return;

    if (document.getElementById('leaflet-css')) {
      initMap();
      return;
    }

    const link = document.createElement('link');
    link.id = 'leaflet-css';
    link.rel = 'stylesheet';
    link.href = 'https://unpkg.com/leaflet@1.9.4/dist/leaflet.css';
    document.head.appendChild(link);

    const script = document.createElement('script');
    script.src = 'https://unpkg.com/leaflet@1.9.4/dist/leaflet.js';
    script.async = true;
    script.onload = () => {
      initMap();
    };
    document.body.appendChild(script);

    return () => {
      if (mapRef.current) {
        mapRef.current.remove();
        mapRef.current = null;
      }
    };
  }, [mounted]);

  // Inicializar mapa de Costa Rica
  const initMap = () => {
    if (!window.L || mapRef.current) return;

    const L = window.L;
    const map = L.map('map-container', {
      center: [9.9281, -84.0907],
      zoom: 9,
      zoomControl: true
    });

    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>',
      maxZoom: 18,
    }).addTo(map);

    mapRef.current = map;
    fetchPlaces();
  };

  // Escuchar cambios en filtros
  useEffect(() => {
    if (mapRef.current) {
      fetchPlaces();
    }
  }, [selectedRegion, selectedCategory]);

  const fetchPlaces = async () => {
    setLoading(true);
    const params = new URLSearchParams();
    if (selectedRegion !== 'Todas') params.append('region', selectedRegion);
    if (selectedCategory !== 'Todas') params.append('category', selectedCategory);

    try {
      const res = await fetch(`/api/places?${params.toString()}`);
      if (!res.ok) throw new Error('Error al consultar la API local');
      
      const data = await res.json();
      if (Array.isArray(data)) {
        setPlaces(data);
        updateMarkers(data);
      }
    } catch (e) {
      console.error('⚠️ [MapAI] Error cargando datos de la API:', e);
    } finally {
      setLoading(false);
    }

    // Re-centrar vista del mapa según región
    if (mapRef.current) {
      if (selectedRegion === 'La Fortuna / San Carlos') mapRef.current.setView([10.4633, -84.7031], 11);
      else if (selectedRegion === 'Manuel Antonio') mapRef.current.setView([9.3892, -84.1383], 12);
      else if (selectedRegion === 'Guanacaste') mapRef.current.setView([10.2993, -85.8402], 10);
      else if (selectedRegion === 'San José') mapRef.current.setView([9.9281, -84.0907], 12);
      else mapRef.current.setView([9.9281, -84.0907], 9);
    }
  };

  const updateMarkers = (items: Place[]) => {
    if (!window.L || !mapRef.current) return;
    const L = window.L;

    // Limpiar marcadores existentes
    markersRef.current.forEach((m) => m.remove());
    markersRef.current = [];

    items.forEach((place) => {
      let lat = 9.9331;
      let lng = -84.0768;

      if (place.location && place.location.coordinates) {
        lng = place.location.coordinates[0];
        lat = place.location.coordinates[1];
      }

      const marker = L.marker([lat, lng]).addTo(mapRef.current);
      
      const popupContent = `
        <div style="font-family: sans-serif; padding: 4px;">
          <strong style="font-size: 14px; color: #0f172a;">${place.name}</strong>
          <p style="font-size: 11px; color: #475569; margin-top: 2px;">${place.short_description}</p>
        </div>
      `;
      marker.bindPopup(popupContent);
      
      marker.on('click', () => {
        setSelectedPlace(place);
      });

      markersRef.current.push(marker);
    });
  };

  if (!mounted) {
    return (
      <div className="w-full h-screen bg-slate-950 flex flex-col items-center justify-center text-emerald-400 font-sans gap-3">
        <div className="w-8 h-8 border-4 border-emerald-500 border-t-transparent rounded-full animate-spin" />
        <p className="text-sm">Iniciando MapAI Costa Rica...</p>
      </div>
    );
  }

  return (
    <div className="relative w-full h-screen flex flex-col md:flex-row overflow-hidden bg-slate-950 text-white font-sans">
      {/* PANEL LATERAL DE NAVEGACIÓN Y GUÍA VIRTUAL */}
      <div className="w-full md:w-96 z-20 bg-slate-900/95 backdrop-blur border-b md:border-r border-slate-800 p-4 flex flex-col gap-4 overflow-y-auto max-h-[45vh] md:max-h-full shadow-2xl">
        <div className="flex justify-between items-center">
          <div>
            <h1 className="text-xl font-bold flex items-center gap-2 text-emerald-400">
              🧭 MapAI Costa Rica
            </h1>
            <p className="text-xs text-slate-400">Exploración turística e historias por IA</p>
          </div>
          {loading && (
            <div className="w-4 h-4 border-2 border-emerald-400 border-t-transparent rounded-full animate-spin" />
          )}
        </div>

        {/* FILTROS DE REGIÓN */}
        <div className="flex flex-col gap-1">
          <label className="text-xs font-semibold text-slate-300">Región</label>
          <select
            value={selectedRegion}
            onChange={(e) => setSelectedRegion(e.target.value)}
            className="bg-slate-800 text-sm border border-slate-700 rounded-lg p-2 focus:ring-2 focus:ring-emerald-500 outline-none text-slate-200"
          >
            {regions.map((r) => (
              <option key={r} value={r}>{r}</option>
            ))}
          </select>
        </div>

        {/* FILTROS DE CATEGORÍA */}
        <div className="flex flex-col gap-1">
          <label className="text-xs font-semibold text-slate-300">Categoría</label>
          <select
            value={selectedCategory}
            onChange={(e) => setSelectedCategory(e.target.value)}
            className="bg-slate-800 text-sm border border-slate-700 rounded-lg p-2 focus:ring-2 focus:ring-emerald-500 outline-none text-slate-200"
          >
            {categories.map((c) => (
              <option key={c} value={c}>{c}</option>
            ))}
          </select>
        </div>

        {/* TARJETA DE LUGAR SELECCIONADO / GUÍA VIRTUAL */}
        {selectedPlace ? (
          <div className="bg-slate-800/90 rounded-xl p-4 border border-emerald-500/40 flex flex-col gap-2 mt-2 shadow-lg animate-fadeIn">
            <div className="flex justify-between items-start">
              <h2 className="font-bold text-lg text-emerald-300">{selectedPlace.name}</h2>
              <span className="text-xs bg-emerald-500/20 text-emerald-400 px-2 py-0.5 rounded-full border border-emerald-500/30">
                {selectedPlace.category}
              </span>
            </div>
            <p className="text-xs text-slate-300 leading-relaxed">{selectedPlace.short_description}</p>
            
            <div className="bg-slate-900/90 p-3 rounded-lg border border-slate-700/60 mt-1">
              <span className="text-[10px] uppercase font-bold text-emerald-400 flex items-center gap-1 mb-1">
                ✨ Guía Virtual IA
              </span>
              <p className="text-xs text-slate-300 italic leading-relaxed">{selectedPlace.ai_story}</p>
            </div>

            <div className="flex justify-between items-center text-xs text-slate-400 mt-2">
              <span>💲 Est. ${selectedPlace.estimated_price_usd} USD</span>
              <span>⭐ {selectedPlace.rating || 4.8}</span>
            </div>
          </div>
        ) : (
          <div className="text-xs text-slate-500 text-center py-8 border border-dashed border-slate-800 rounded-xl flex flex-col items-center gap-2">
            <span>📍 Haz clic en cualquier marcador del mapa para desplegar los detalles e historia de la Guía Virtual.</span>
          </div>
        )}
      </div>

      {/* CONTENEDOR DEL MAPA */}
      <div className="flex-1 w-full h-full relative z-10 bg-slate-950">
        <div id="map-container" className="w-full h-full" />
      </div>
    </div>
  );
}

declare global {
  interface Window {
    L: any;
  }
}
